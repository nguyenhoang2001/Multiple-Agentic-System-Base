# Smart Home Automation Rules
# Available devices: light_living_01 (living room), light_bedroom_01 (bedroom), thermostat_01 (bedroom)
# Format: rule_id | trigger | condition | action | priority

rule_id: R001
name: Morning Wake-Up
trigger: schedule at 07:00 on weekdays
condition: none
action: set light_bedroom_01 brightness=30%, set thermostat_01 setpoint=22°C mode=heat
priority: high
description: Gently turn on the bedroom light and warm the room before the occupant wakes.

rule_id: R002
name: Night Mode
trigger: schedule at 23:00
condition: none
action: turn_off light_living_01, turn_off light_bedroom_01, set thermostat_01 setpoint=19°C mode=auto
priority: high
description: Turn off all lights and lower thermostat for nighttime.

rule_id: R003
name: Overheating Alert
trigger: thermostat_01 temperature > 30°C
condition: thermostat_01 mode != cool
action: set thermostat_01 mode=cool setpoint=24°C, send_notification "High temperature alert: {value}°C"
priority: critical
description: Switch to cooling and alert the user when temperature exceeds safe threshold.

rule_id: R004
name: Cold Alert
trigger: thermostat_01 temperature < 14°C
condition: thermostat_01 mode != heat
action: set thermostat_01 mode=heat setpoint=20°C, send_notification "Low temperature alert: {value}°C"
priority: critical
description: Switch to heating when temperature drops too low.
